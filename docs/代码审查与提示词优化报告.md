## 一、总体印象

当前代码已经形成了一条**完整的数据→研判→报告→回测管线**，且项目记忆里的“不变式”执行得较好（颜色约定、章节结构、7只自选股、实时数据驱动、入库、外网英文、微博解构等）。主要问题集中在：

1. **代码重复与硬编码**（美股代码映射、ETF列表、自选股名称、词库等）。
2. **提示词/研判文本零散内联**，缺乏统一模板与版本管理。
3. **报告生成模块职责过重**，`build_report.py` 同时承担数据加载、研判、HTML渲染、数据库存储。
4. **LLM 相关提示词较原始**（多空词库、否定词、外网总结等），可升级为结构化 Prompt + 示例驱动的系统提示。
5. **错误处理与日志**可更统一，部分模块“静默失败”风险仍存在。
6. **数据库连接与性能**可优化（连接池、批量插入、初始化合并）。

1. **网络状态缓存加 TTL**

```python
import time

_NETWORK_OK = None
_NETWORK_TIME = 0
NETWORK_TTL = 30  # 秒

def check_network(host="news.google.com", port=443, timeout=4):
    global _NETWORK_OK, _NETWORK_TIME
    if _NETWORK_OK is not None and time.time() - _NETWORK_TIME < NETWORK_TTL:
        return _NETWORK_OK
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(timeout)
        s.connect((host, port))
        s.close()
        _NETWORK_OK, _NETWORK_TIME = True, time.time()
    except Exception:
        _NETWORK_OK, _NETWORK_TIME = False, time.time()
    return _NETWORK_OK

@functools.lru_cache(maxsize=1)
def collect_depth():
    return "deep" if check_network() else "shallow"
```

```python
def analyze_sentiment(weibo_data, quotes):
    all_text = " ".join(p["text"] for posts in weibo_data.values() for p in posts)
    matched_sectors = extract_keywords(all_text, SECTOR_KEYWORDS)

    chg_pcts = [q["chg_pct"] for q in quotes.values() if "chg_pct" in q]

    up_count = sum(1 for c in chg_pcts if c > 0)
    strong_up = sum(1 for c in chg_pcts if c > 1.5)
    strong_down = sum(1 for c in chg_pcts if c < -1.5)

    if strong_up >= 3 and avg_chg > 1.0:
        market_state = "bullish"
    elif strong_down >= 3 and avg_chg < -1.0:
        market_state = "bearish"
    elif abs(avg_chg) <= 0.5 and up_count in [2, 3]:
        market_state = "neutral"
    elif avg_chg > 0:
        market_state = "bullish"
    else:
        market_state = "bearish"

    return {
        "matched_sectors": matched_sectors,
        "market_state": market_state,
        "up_count": up_count,
        "avg_chg": round(avg_chg, 2),
        "total_indices": len(quotes),
    }
```

4. **`fetch_url` 增加更完整的请求头**

```python
def fetch_url(url, headers=None, timeout=10, quiet=False, encoding="utf-8"):
    req = urllib.request.Request(url)
    req.add_header("User-Agent", UA)
    req.add_header("Accept", "*/*")
    req.add_header("Accept-Language", "zh-CN,zh;q=0.9,en;q=0.8")
    req.add_header("Accept-Encoding", "gzip, deflate, br")
    req.add_header("Connection", "keep-alive")
    if headers:
        for k, v in headers.items():
            req.add_header(k, v)
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            data = resp.read()
            # 处理 gzip
            import gzip
            if resp.headers.get("Content-Encoding") == "gzip":
                data = gzip.decompress(data)
            return data.decode(encoding, errors="replace")
    except Exception as e:
        if not quiet:
            print(f"  [fetch error] {url[:80]} -> {e}")
        return None
```

```python
import jieba

def _score_text(text: str) -> float:
    """基于分词、否定范围的多空打分。"""
    if not text:
        return 0.0
    tokens = list(jieba.cut(text))
    score = 0.0
    neg_window = 3
    for i, tok in enumerate(tokens):
        if tok in BULL_WORDS:
            window = tokens[max(0, i - neg_window):i]
            mul = -1.0 if any(w in NEG_WORDS for w in window) else 1.0
            score += mul
        elif tok in BEAR_WORDS:
            window = tokens[max(0, i - neg_window):i]
            mul = -1.0 if any(w in NEG_WORDS for w in window) else 1.0
            score -= mul
    return score
```

```python
def init_database(self):
    conn = psycopg2.connect(**self.conn_params, dbname="postgres")
    conn.autocommit = True
    try:
        with conn.cursor() as cur:
            cur.execute("SELECT 1 FROM pg_database WHERE datname = %s", (self.dbname,))
            if not cur.fetchone():
                cur.execute(_sql.SQL('CREATE DATABASE {}').format(_sql.Identifier(self.dbname)))
    finally:
        conn.close()

    conn = psycopg2.connect(**self.conn_params, dbname=self.dbname)
```

2. **引入连接池**

```python
from psycopg2 import pool

class StockAgentDB:
    def __init__(self, ...):
        self.conn_params = dict(host=host, port=port, user=user, password=password)
        self.dbname = dbname
        self._pool = None

    def _pool(self):
        if self._pool is None:
            self._pool = pool.ThreadedConnectionPool(
                minconn=1, maxconn=5,
                dbname=self.dbname, **self.conn_params
            )
        return self._pool

    @contextmanager
    def _cursor(self):
        conn = self._pool().getconn()
        try:
            cur = conn.cursor()
            try:
                yield cur
                conn.commit()
            except Exception:
                conn.rollback()
                raise
            finally:
                cur.close()
        finally:
            self._pool().putconn(conn)
```

```python
_NEGATORS = ["not", "no", "unlikely", "don't expect", "ruled out"]

def _classify_stance(text):
    low = text.lower()
    sentences = re.split(r"[.!?]", low)
    hawk, dove = 0, 0
    for s in sentences:
        neg = any(n in s for n in _NEGATORS)
        h = sum(1 for k in _HAWKISH if re.search(rf"\b{re.escape(k)}\b", s))
        d = sum(1 for k in _DOVISH if re.search(rf"\b{re.escape(k)}\b", s))
        if neg:
            h, d = d, h
        hawk += h
        dove += d
    if hawk > dove + 1:
        return "偏鹰"
    if dove > hawk + 1:
        return "偏鸽"
    return "中性"
```
